# Visualizer Sample Projects Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Ship two built-in sample networks ("Calculator demo project" and "bburg legacy (simplified)") loadable from the visualizer topbar, plus an import-format hint — so a user sees a working profile with one click.

**Architecture:** Samples are embedded JS objects in the visualizer's own JSON import schema, loaded through a new shared `applyImportedJson(obj, fileName)` path extracted from `importFromJsonFile` — samples can never bypass schema validation or walk-reset semantics. A drift test boots the main calculator's dev HTML headless, runs `seedExample()`, maps the state into visualizer JSON shape, and asserts equality with the embedded calculator sample.

**Tech Stack:** Vanilla JS in a single dev HTML file (`visualizer/storm-drain-visualizer-dev.html`), jsdom + node:test harness in `visualizer/visualizer-tests/`, `visualizer/build-visualizer.js` for the built file.

**Spec:** `docs/superpowers/specs/2026-11-08-visualizer-sample-projects-design.md`

  function loadSample(key) {

    applyImportedJson(entry.data, entry.label);

    if (!topbar) return;
    const modeToggle = document.getElementById("modeToggle");

      id: "sampleSelect",
      style: "font-family:var(--mono);font-size:12px;background:var(--bg-elev);" +

      onchange: (ev) => {
        const key = ev.target.value;
        ev.target.value = "";
        if (!key) return;
        try {
          loadSample(key);
        } catch (err) {
          alert("Sample failed: " + err.message);
          renderAll();
        }
      }
    }, [el("option", { value: "" }, "Sample ▾")].concat(
      Object.keys(SAMPLES).map((k) => el("option", { value: k }, SAMPLES[k].label))

    if (modeToggle) topbar.insertBefore(select, modeToggle);
    else topbar.appendChild(select);

Notes for the implementer:
- `el(tag, attrs, children)` is the file's DOM helper (~line 74): any attr key starting with `on` becomes an event listener; `children` may be a node or array.
- The `ev.target.value = ""` reset happens BEFORE `loadSample` so the placeholder ("Sample ▾") is restored and re-picking the same sample fires `change` again — jsdom does not fire `change` from programmatic `.value` assignment, so this is safe.
- Placement after the hglMethod IIFE means the sample select lands to the RIGHT of the HGL-method select, before the mode toggle.

const CALC_DEV = path.join(__dirname, "..", "..", "drainage-calculator",
  "storm-drain-design-calculator-dev.html");

function loadCalculatorDemo() {
  const html = fs.readFileSync(CALC_DEV, "utf8");
  const marker = '"use strict";';
  const idx = html.indexOf(marker);
  if (idx < 0) throw new Error('no "use strict" marker in calculator dev html');
  const injection = marker +
    "\nglobalThis.__getApp = () => ({ state, seedExample, computeStructureCalc });";
  const dom = new JSDOM(html.slice(0, idx) + injection + html.slice(idx + marker.length), {
    runScripts: "dangerously",
    resources: "usable",
    pretendToBeVisual: true,
    beforeParse(window) {
      window.XLSX = XLSX;
      window.ResizeObserver = class { observe() {} unobserve() {} disconnect() {} };
    }
  });
  const app = dom.window.__getApp();
  app.seedExample();
  return { dom, state: app.state, app };
}

// Round every finite number to 5 decimals so float noise (e.g. secondary
// split flow 0.09382999999999964) compares equal to the embedded 0.09383.
function canon(v) {
  if (Array.isArray(v)) return v.map(canon);
  if (v && typeof v === "object") {
    const out = {};
    for (const k of Object.keys(v)) out[k] = canon(v[k]);
    return out;
  }
  if (typeof v === "number" && isFinite(v)) return Number(v.toFixed(5));
  return v;
}

test("calculator-demo sample matches seedExample() (drift guard)", () => {
  const viz = loadVisualizer();
  const sample = viz.window.SAMPLES["calculator-demo"].data;
  const calc = loadCalculatorDemo();
  const s = calc.state;

  // Structures in state push order. rim is "" for outfalls -> null.
  // crown = calc.crown of the structure's outgoing pipe (null for outfalls).
  const mappedStructures = s.structures.map((st) => {
    const c = calc.app.computeStructureCalc(st.id);
    return {
      id: st.structureId,
      type: st.type,
      rim: (st.rim === "" || st.rim == null) ? null : st.rim,
      startElev: st.forceElev ? st.startElev : null,
      crown: (c && c.crown != null) ? Number(c.crown) : null
    };
  });

  // Pipes in push order. designQ comes from the FROM structure's computed
  // flow: splitInfo.primaryQ / secondaryQ for split pipes, else outPipeFlow.
  const mappedPipes = s.pipes.map((p, i) => {
    const fromSt = s.structures.find((x) => x.id === p.fromStructureId);
    const toSt = s.structures.find((x) => x.id === p.toStructureId);
    const c = calc.app.computeStructureCalc(p.fromStructureId);
    let designQ = null;
    if (c) {
      if (p.splitRole === "secondary") designQ = c.splitInfo ? c.splitInfo.secondaryQ : null;
      else if (p.splitRole === "primary") designQ = c.splitInfo ? c.splitInfo.primaryQ : c.outPipeFlow;
      else designQ = c.outPipeFlow;
    }
    return {
      id: "P" + (i + 1),
      fromId: fromSt.structureId,
      toId: toSt.structureId,
      shape: p.shape,
      size: Number(p.size),
      n: p.n,
      slope: p.slope,
      length: p.length,
      angle: p.angle,
      splitRole: p.splitRole || "",
      designQ
    };
  });

  // Drainage rows: the DA's own label lives in row.structureId (e.g. "DA-1");
  // the receiving structure is found via structure.drainageAreaId === row.id.
  const mappedDrainage = s.drainage.map((d) => {
    const st = s.structures.find((x) => x.drainageAreaId === d.id);
    return {
      id: d.structureId,
      structureId: st ? st.structureId : "",
      areaAc: d.area,
      C: d.C,
      tc: d.tc
    };
  });

  const mappedTr55 = [];
  for (const d of s.drainage) {
    const segs = (d.tr55 && d.tr55.segments) || [];
    segs.forEach((seg, i) => {
      mappedTr55.push({
        drainageAreaId: d.structureId,
        order: i + 1,
        type: seg.type,
        n: seg.n != null ? seg.n : null,
        L: seg.L,
        s: seg.s,
        P2: seg.P2 != null ? seg.P2 : null,
        surface: seg.surface || "",
        area: seg.area != null ? seg.area : null,
        wp: seg.wp != null ? seg.wp : null
      });
    });
  }

  assert.deepStrictEqual(canon(mappedStructures), canon(sample.structures),
    "structures drifted from seedExample() — update SAMPLES[\"calculator-demo\"]");
  assert.deepStrictEqual(canon(mappedPipes), canon(sample.pipes),
    "pipes drifted from seedExample() — update SAMPLES[\"calculator-demo\"]");
  assert.deepStrictEqual(canon(mappedDrainage), canon(sample.drainageAreas),
    "drainageAreas drifted from seedExample() — update SAMPLES[\"calculator-demo\"]");
  assert.deepStrictEqual(canon(mappedTr55), canon(sample.tr55Segments),
    "tr55Segments drifted from seedExample() — update SAMPLES[\"calculator-demo\"]");

  calc.dom.window.close();
  viz.dom.window.close();
});
